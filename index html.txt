  function startLesson(e) {
    e.preventDefault();
    document.getElementById('loginPage').style.display = 'none';
    document.getElementById('lessonPage').style.display = 'block';
  }

  function changeLevel() {
    const level = document.getElementById('level').value;
    const text = document.getElementById('lessonText');

    if (level === 'easy') {
      text.innerText = "Let's learn very simply 😊. Imagine you are playing and learning at the same time. Real life is our best teacher.";
    } else if (level === 'medium') {
      text.innerText = "Now let's learn step by step with examples you see every day, like school and home.";
    } else {
      text.innerText = "Let's go deeper into the topic with detailed explanation and logic.";
    }
  }

  function aiExplain() {
    const question = document.querySelector('textarea').value;
    const level = document.getElementById('level').value;
    const text = document.getElementById('lessonText');

    if (!question) {
      alert('Please write a question or chapter first');
      return;
    }

    text.innerText = "🤖 Miss Wisdom is thinking...";

    setTimeout(() => {
      if (level === 'easy') {
        text.innerText = "😊 Easy explanation (like a human teacher): " + question + " is explained like a story. Imagine it in daily life, just like learning while playing.";
      } else if (level === 'medium') {
        text.innerText = "📘 Medium explanation: " + question + " is explained step by step with school and home examples.";
      } else {
        text.innerText = "📗 Hard explanation: " + question + " is explained deeply with logic, reasons, and real-world connections.";
      }
    }, 1500);
  }
